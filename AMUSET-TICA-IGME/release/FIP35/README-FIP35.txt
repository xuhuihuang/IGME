Jupyter Notebook

FIP35-tICA-gen.ipynb:
# Loads MD trajectories (sub-sampled every 5 frames; 1 frame = 1 ns after 
# sub-sample) and does tICA with parameters lag_time=10. Saves the top 5 tICs 
# into "FIP35-tICA.npz". Also contains some plotting against different tICs
# but finally none of them came into the draft of paper.

FIP35-Koopman.ipynb:
# Uses tICs to generate Koopman matrices of rank 27 with AMUSET-TICA library 
# and outputs the result into "FIP35-Ks.npy". Contains the loading and usage 
# of file "FIP35-tICA.npz".

FIP35-savings-comparison.ipynb:
# Two parts. 1. Comparing rank 27, 125, 343 for the minimal trajectory length
# needed for ITS reaching gold standard; 2. At rank 27, compare ITS results 
# at different lag times for AMUSET-TICA and A.T.-IGME.

FIP35-ITS-comparison.ipynb:
# Does tICA, AMUSET-TICA, A.T.-IGME at different lag times for demonstrating 
# savings over different methods.

FIP35-RMSE_n_CV.ipynb:
# Goes over hyperparameters tau_k and L for fitting. Creates CV plot against 
# RMSD 1 & 2 for predicted CV with propagation parameters tau_k=42, L=25, 
# at 300ns lag time; saved to "FIP35_pred300lag-mode*_data.txt".

FIP35-foldingPath.ipynb:
# Creates the CVs for tau_k=30 and L=15 and outputs into "CVs_pred30-45.npy". 
# Performs 3D plot for CV1, CV3 and CV4. Plots are reproduced in a nicer way 
# in "FIP35-AllPlots.ipynb".

FIP35-tICsGaussian_n_bootstraping.ipynb:
# Two parts. 1. Bootstrapping for CK test; 2. Plotting the tICs expansions.

FIP35-AllPlots.ipynb:
# Collects the data from all previous codes and generates the plots ready for
# publications. Has color-friendly plans, and designs neat plots.

FIP35-dim27-SVD-Vt.npy:
# Save the V^T matrices from SVD decomposition results which are ready for 
# generating CVs of rank 27. The CVs are generated by np.matmul(Vt, vrK), 
# in which vrK is the sorted eigenvector matrix from diagonalizing Koopman 
# matrix.

FIP35-Ks.npy:
# Koopman matrices generated by AMUSET-TICA for rank 27 at lag times ranging
# from 1ns to 100ns.

FIP35-Ks-343.npy:
# Koopman matrices generated by AMUSET-TICA for rank 343 at lag times ranging
# from 1ns to 100ns.

FIP35-Ks_lag100-300.npy:
# Koopman matrices generated by AMUSET-TICA for rank 27 at lag times in
# range(100, 301, 10).

FIP35-tICA.npz:
# tICs generated using msmbuilder library. Created in FIP35-tICA-gen.ipynb. 
# Check the usage in FIP35-Koopman.ipynb.

rmsd-beta1\, rmsd-beta2\:
# RMSD 1 and 2 for CV plotting. See usage in FIP35-RMSE_n_CV.ipynb.

Koopman_bootstrap.npz:
# CK test intermediate results. Has 3 keys: "Koop_boot" saves a 4D np.array 
# which is the matrices during bootstrapping; "lag" saves the corresponding
# lag times; "note" is the note. See FIP35-tICsGaussian_n_bootstraping.ipynb,
# which is the notebook generating this .npz file, for more information.

random-number-58.txt:
# Random numbers, generated by website random.org. Used in CK test. Check
